# ====================================================================
# PHASE SEQUENCE parameters (orient / search / post_contact_align / surface_impedance)
# ====================================================================
# Everything for the phase-sequence run. Surface + tool geometry are in
# common.txt; the auto-damping toggles/factors are in common.txt. This file
# OWNS all auto-written keys (method2_K/D_tcp, last_chasles_axis_*,
# last_best_axis_*) -- the controller writes them back here after a run.

# use_phase_sequence = 1 runs the controller in this order:
#   1. orient_to_surface     -- rotate to match the virtual surface orientation.
#   2. search_first_contact  -- translate along contact_search_direction until either
#                                force contact is detected or the pre-surface
#                                clearance is reached, depending on contact_search_mode.
#   3. post_contact_align    -- press the contacted edge into the surface
#                                while holding a soft rotational spring, so
#                                real contact moment passively tips the tool flat.
#   4. surface_impedance     -- normal-constrained impedance hold/test.
use_phase_sequence = 1

# 0 = skip orient_to_surface completely. The run starts at search_first_contact
# and holds the q_init/start orientation during search; after contact it still
# enters post_contact_align because use_phase_sequence remains 1.
use_orientation_phase = 1

# ====================================================================
# Phase 1: orient_to_surface
# ====================================================================
# (Surface orientation / tilt angles are defined in common.txt.)

# Minimum time before switching to search [s].
orient_phase_min_time = 0.5

# Orientation error threshold for switching to search [rad].
# Smaller = aligns more accurately before switching, but the orient gains below
# must be stiff enough that the steady-state error actually drops this low,
# otherwise the phase never switches. 0.035 rad ~ 2 deg.
orient_phase_error_threshold = 0.035

# ----- Shared "approach" impedance (orient_to_surface + search_first_contact) -----
# One translation stiffness + one rotation stiffness for BOTH early phases,
# alignment-target frame [tangent1, tangent2, normal]. Position soft (gentle
# search contact; orient holds loosely); KR_tangent stiff enough that orient
# converges below orient_phase_error_threshold.
# (Auto-damping toggle/factor: approach_auto_damping[_factor] in common.txt.)
approach_Kp_normal = 150.0
approach_Kp_tangent1 = 150.0
approach_Kp_tangent2 = 150.0
approach_KR_normal = 8.0
approach_KR_tangent1 = 90.0
approach_KR_tangent2 = 90.0
# Manual damping FALLBACK [Ns/m | Nms/rad], paired with the approach K gains.
# Used only when approach_auto_damping is off, OR when it is on but the online
# task-space inertia estimate is unavailable that cycle -- i.e. near a kinematic
# singularity (Jacobian loses rank) or if the inertia comes out non-finite /
# non-positive. Otherwise D = factor*2*sqrt(M*K) is computed live and these are
# ignored.
approach_Dp_normal = 20.0
approach_Dp_tangent1 = 20.0
approach_Dp_tangent2 = 20.0
approach_DR_normal = 12.0
approach_DR_tangent1 = 12.0
approach_DR_tangent2 = 12.0

# 1 = test only the rotation part (hold TCP, rotate toward R_d_alignment_target, skip
# contact search). Set back to 0 once orientation control is verified.
orientation_test_only = 0

# Extra test rotation around surface tangent1 [deg], applied to the start
# orientation. Only used when orientation_test_only = 1.
orientation_test_extra_tilt_deg = 0.0

# Force safety trigger during orient. NOTE: only active in FORCE search mode --
# in pre_surface (geometric) mode it is bypassed, so the orient-correction
# transient cannot be misread as contact.
detect_contact_during_alignment = 1
alignment_contact_force_threshold = 12.0

# ====================================================================
# Phase 2: search_first_contact
# ====================================================================
# TCP moves slowly in contact_search_direction. contact_search_mode selects
# the switch trigger:
#   force       = old behavior, switch after confirmed external-force threshold.
#   pre_surface = no force comparison, switch when the active tool contact point
#                 is contact_search_surface_clearance before the configured
#                 surface plane.
use_contact_search = 1
contact_search_mode = pre_surface
contact_search_surface_clearance = 0.020

# 1 = search along -alignment_target_normal. 0 = use the manual direction below.
contact_search_use_alignment_target_normal = 1
contact_search_direction_x = 0.0
contact_search_direction_y = 0.0
contact_search_direction_z = -1.0

# Search speed [m/s] and maximum travel distance [m].
contact_search_speed = 0.1
contact_search_max_distance = 0.640

# Minimum downward travel before first-touch contact detection [m].
contact_search_min_distance = 0.240

# First-touch reference/gain calculation uses the same gate as the phase switch.
contact_search_first_touch_min_distance = 0.240

# First-touch and phase switch use dF_bias dot contact_search_direction when enabled.
contact_search_use_directional_force = 1
contact_search_confirm_time = 0.0
contact_force_threshold = 12.0

# Cartesian gains during contact search come from the shared "approach" block
# in Phase 1 (approach_Kp/KR/Dp/DR) -- the same set orient uses.

# 1 = once contact is found, target a tool orientation parallel to the
# surface (for the horizontal table, tool z points down along -base z).
align_orientation_to_surface_after_contact = 1

# ====================================================================
# Phase 3: post_contact_align
# ====================================================================
# Minimum time before the moment-threshold early exit can fire [s].
post_contact_align_min_time = 2.0

# Maximum time spent in this phase after first contact [s].
post_contact_align_duration = 3.0

# Moment-change early-exit threshold [Nm] (e.g. the second tool edge
# touching down after passively tipping flat). Only checked once
# post_contact_align_min_time has elapsed.
post_contact_moment_threshold = 150.0

# Ignore the first transient screw-axis samples when picking the measured
# "best" axis for comparison/gain suggestion; among stable samples, picks
# the one closest to the active contact edge. If none pass the gate, the
# best transient sample is kept and labeled transient_fallback.
post_contact_best_axis_min_time = 0.60
post_contact_best_axis_min_omega = 0.12

# Small position offset into the real surface [m], acting like a spring
# preload that presses the selected edge down:
#   preload force approx post_contact_Kp_z * post_contact_normal_push
# Force-search mode starts the push ramp at post_contact_normal_push; pre-surface
# mode starts it at -contact_search_surface_clearance. BOTH now use the single
# post_contact_Kp_z (no separate pre-surface stiffness).
# Check peak preload force (post_contact_Kp_z * post_contact_max_push) against
# collision_force_acc/nom (safety.txt) whenever Kp_z or max_push changes.
# This preload is also the one grind inherits (post_contact_hold_push), so it
# sets the press for BOTH the align tip and the grind sweep.
post_contact_normal_push = 0.250
post_contact_push_speed = 0.050
post_contact_max_push = 0.180

# Disable the surface-impedance phase. 1 = when post_contact_align ends (moment
# threshold or post_contact_align_duration), freeze the preload at its current
# value and keep post_contact_align running as a constant-press, free-slide
# hold: the tool keeps pressing into the table with that preload (only the
# normal direction is sprung) but is tangentially free, so it can be moved along
# the table by hand; rotation stays held by post_contact_KR. 0 = legacy: hand off
# to surface impedance.
post_contact_hold_after_align = 1

# Phase-sequence Enter gates (press a bare Enter to continue; e+Enter stops).
#   pause_before_align = 1 -> hold at the pre-surface point (surface_clearance
#     above the plane) and wait for Enter before starting the alignment press.
#   pause_after_align = 1 -> after post_contact_align finishes, hold the pressed
#     pose and wait for Enter before starting the grind/hold.
pause_before_align = 1
pause_after_align = 0
# Stiff hold while paused at Gate A so the tool locks in place (resists being
# pushed by hand) instead of holding with the soft search gains. Centered on the
# tool's actual position at the moment the gate arms. POSITION only -- rotation
# stays on the approach spring at the gate (a stiff isotropic rotational lock
# here drove a limit-cycle wiggle, so it was removed).
gate_hold_Kp = 5000.0
gate_hold_Dp = 200.0

# Translational spring used only during this phase, base frame [x, y, z].
# Firm normal (z) for press + SOFT tangents (x,y) so the tool is free to rotate
# about the edge (stiff tangents would pin the TCP and shove the arm up instead
# of tipping the tool).
post_contact_Kp_x = 2000.0
post_contact_Kp_y = 2000.0
post_contact_Kp_z = 350.0
# Manual damping FALLBACK [Ns/m], paired with the post_contact_Kp gains above.
# Used only when post_contact_auto_damping is off, OR when it is on but the
# online task-space inertia estimate is unavailable that cycle (kinematic
# singularity / non-finite / non-positive inertia). Otherwise D =
# post_contact_auto_damping_factor*2*sqrt(M*K) is computed live.
post_contact_Dp_x = 10.0
post_contact_Dp_y = 10.0
post_contact_Dp_z = 175.0

# Rotational spring used only during this phase, alignment-target frame
# [tangent1, tangent2, normal]. Target is the pose held at first contact (no
# scripted rotation). Normal (yaw about the target axis) stays constrained;
# tangent1/tangent2 (the tipping directions) stay very soft so real contact
# moment can passively rotate the tool flat.
post_contact_KR_normal = 10.0
post_contact_KR_tangent1 = 0.05
post_contact_KR_tangent2 = 0.05
# Manual damping FALLBACK [Nms/rad], paired with the post_contact_KR gains above
# (same fallback conditions as post_contact_Dp).
post_contact_DR_normal = 10.0
post_contact_DR_tangent1 = 10.01
post_contact_DR_tangent2 = 10.01

# --------------------------------------------------------------
