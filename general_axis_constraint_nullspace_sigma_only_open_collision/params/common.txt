# --------------------------------------------------------------------
# Auto-damping behaviour (toggles + factors, shared here)
# --------------------------------------------------------------------
# Each phase can compute its damping online as D = factor * 2 * sqrt(M * K),
# with M the libfranka task-space inertia and K that phase's stiffness. These
# toggles/factors control that behaviour; the per-phase manual Dp/DR FALLBACK
# values stay next to their K gains in sequence.txt / hold.txt and are only used
# when the toggle is 0, or when the online inertia estimate is unavailable that
# cycle (see the fallback note there).
# factor: 1.0 = critical, <1 lighter, >1 heavier.

# Hold mode: compute D once at hold entry from hold_Kp.
hold_auto_damping = 1
hold_auto_damping_factor = 1.0

# orient_to_surface + search_first_contact (the shared "approach" gains).
# factor 1.9 ~ matches the old flat approach_DR_tangent (~12) on the orient DOF.
approach_auto_damping = 1
approach_auto_damping_factor = 1.9

# post_contact_align + grind. Position uses base-frame inertia, rotation uses the
# alignment-target-frame inertia. factor 0.8 ~ matches the old post_contact_Dp_z.
post_contact_auto_damping = 1
post_contact_auto_damping_factor = 0.8

# --------------------------------------------------------------------
# Gripper actions (used by both sequence and guidance runs)
# --------------------------------------------------------------------
# Perform a Franka hand action after q_init and before contact search, as late
# as possible before contact. require_gripper_open = 1 stops the run if that
# action fails; = 0 makes it non-blocking so the run proceeds regardless.

require_gripper_open = 0

# Which action to perform:
#   gripper_grasp_on_tool = 0 -> OPEN the hand to gripper_open_width (releases).
#   gripper_grasp_on_tool = 1 -> CLOSE/grasp onto the tool you are holding:
#       drive the fingers to gripper_grasp_width and clamp with
#       gripper_grasp_force. Set gripper_grasp_width to your tool's thickness at
#       the grip point (measure it); the grasp succeeds if the final width is
#       within [width - epsilon_inner, width + epsilon_outer]. On a re-run it
#       detects the tool is already held and keeps the grasp (no drop).

gripper_grasp_force = 60.0

# surface_point_z is the plane height AT THE TOOL directly -- decoupled from the
# tilt. Changing the tilt now pivots the plane about this point WITHOUT moving
# its height at the tool, so you set the angle and the height independently.

# Surface ORIENTATION (the plane normal). Two ways to set it:
#   use_alignment_target_tilt_angle = 1 -> compute the normal from two tilt
#     angles: n = R_y(b)*R_x(a)*[0,0,1] = [sin(b)cos(a), -sin(a), cos(b)cos(a)].
#     a = tilt about base x, b = tilt about base y. a = b = 0 = horizontal.
#   use_alignment_target_tilt_angle = 0 -> use the manual normal vector below.

# --------------------------------------------------------------------
# Nullspace sigma-min optimization
# --------------------------------------------------------------------
# Uses the 1D nullspace direction n of the 6x7 Jacobian: compares
# J(q + alpha*n) with J(q - alpha*n) and prefers the direction with larger
# sigma_min(J), while a restoring spring keeps q close to the start posture.
# nullspace_mode:
#   0/off     = no nullspace torque
#   1/posture = tau_nullspace only: N * (k_start*(q_start-q) - d*dq)
#   2/sigma   = tau_sigma only: singular-value direction
#   3/both    = old combined behavior

nullspace_damping = 2.0
nullspace_k_start = 2.0
nullspace_k_sigma = 0.1
