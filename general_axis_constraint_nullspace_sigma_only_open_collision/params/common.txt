# --------------------------------------------------------------------
# Auto-damping behaviour (toggles + factors, shared here)
# --------------------------------------------------------------------
# Each phase group can compute its damping online as D = factor*2*sqrt(M*K),
# with M the libfranka task-space inertia and K that group's stiffness. The
# per-group manual Dp/DR FALLBACK values stay next to their K gains in
# sequence.txt / hold.txt, and are used only when the toggle here is 0, or when
# it is 1 but the online inertia estimate is unavailable (see the note there).
# factor: 1.0 = critical, <1 lighter, >1 heavier.

# Hold mode: computed once at hold entry from hold_Kp.
hold_auto_damping = 1
hold_auto_damping_factor = 1.0

# Phase 1, both approach steps (they share one gain set).
# factor 1.9 ~ matches the old flat approach_DR_tangent (~12) on the orient DOF.
approach_auto_damping = 1
approach_auto_damping_factor = 1.9

# Phases 2 and 3 (set up + grind, they share one gain set). Position uses
# base-frame inertia, rotation uses the alignment-target-frame inertia.
# factor 0.8 ~ matches the old setup_Dp_z.
setup_auto_damping = 1
setup_auto_damping_factor = 0.8

# Upper clamp on any auto-damping value computed from the formula above.
auto_damping_max = 8000.0

# How the manual Dp/DR entries relate to the computed value:
#   0 = pure FALLBACK. The computed value is used as-is; the manual entry only
#       steps in when the inertia estimate is unavailable that cycle.
#   1 = the manual entry is also a per-axis FLOOR: D_i = max(computed_i,
#       manual_i).
#
# This matters on very soft axes. D = factor*2*sqrt(M*K) goes to zero as K does,
# so an axis with almost no stiffness ends up with almost no damping -- most
# visibly on the set-up tipping axes, where setup_KR_tangent1/2 = 0.05 puts the
# computed DR far below the setup_DR_tangent1/2 entry.
#
# Turning this on makes the tool tip noticeably slower during set up, which is
# the core motion of the experiment -- so switch it on deliberately, re-run, and
# compare. Leave the print below on to see both numbers first.
auto_damping_min_from_manual = 0

# 1 = print the computed damping next to the manual entry whenever a phase
# group's auto-damping is (re)computed.
print_auto_damping = 1

# --------------------------------------------------------------------
# Nullspace sigma-min optimization
# --------------------------------------------------------------------
# Uses the 1D nullspace direction n of the 6x7 Jacobian: compares J(q + alpha*n)
# with J(q - alpha*n) and prefers the direction with the larger sigma_min(J),
# while a restoring spring keeps q close to the start posture. Always off during
# the approach, where the only job is to move the TCP toward the plane.
# nullspace_mode:
#   0/off     = no nullspace torque
#   1/posture = tau_nullspace only: N * (k_start*(q_start-q) - d*dq)
#   2/sigma   = tau_sigma only: the singular-value direction
#   3/both    = combined

nullspace_mode = 3
nullspace_damping = 2.0
nullspace_k_start = 2.0
nullspace_k_sigma = 0.1
nullspace_alpha = 0.01
