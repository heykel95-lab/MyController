# ------------------------------------------------------------------
# Surface constraint mode
# ------------------------------------------------------------------
# constraint_enabled = 1:
#
# The constrained surface is a plane:
#
#   n^T * (p - p_surface) = 0
#
# surface_normal defines n in the robot base frame.
# use_start_as_surface_point = 1 uses the start TCP position as p_surface.
# This is safest because the surface is never far from the initial robot pose.
#
# The impedance gains are chosen in the surface frame:
#
#   normal   = surface normal direction
#   tangent1 = first tangent direction
#   tangent2 = second tangent direction
#
# Then the code transforms them to the robot base frame:
#
#   K_base = R_surface * K_surface * R_surface^T
#
# The same mode covers axis-aligned and inclined planes:
#   Y-Z plane: surface_normal = [1, 0, 0]
#   X-Z plane: surface_normal = [0, 1, 0]
#   X-Y plane: surface_normal = [0, 0, 1]
#
# Inclined surfaces use the same mode with an inclined normal.
#
# Current example, horizontal table:
#   surface_normal = [0, 0, 1]
#
# Old Y-Z plane example:
#   surface_normal = [1, 0, 0]

# ------------------------------------------------------------------
# Phase sequence for tilted-tool alignment
# ------------------------------------------------------------------
# use_phase_sequence = 1 runs the controller in this order:
#   1. orient_to_surface:
#      rotate the tool at the start position until R_EE is parallel to
#      the virtual surface orientation R_d_surface.
#   2. search_first_contact:
#      translate in contact_search_direction while keeping R_EE near R_d_surface.
#   3. post_contact_align:
#      after first contact, keep pressing the contacted edge into the
#      surface (stiff downward push) while holding a very soft rotational
#      spring, so a real contact moment can passively tip the tool flat
#      instead of the controller scripting the rotation.
#   4. surface_impedance:
#      normal constrained impedance experiment.
use_phase_sequence = 1

# Minimum time spent in orient_to_surface before switching [s].

# Orientation error threshold for switching from orient_to_surface to search [rad].

# Minimum time before moment comparison can end post_contact_align [s].

# Maximum time spent in post_contact_align after first contact [s].
# The no-sideways test settled by about 2-3 s; 8 s keeps a larger safety margin
# without holding the high preload for the old 15 s window.
post_contact_align_duration = 8.0

# Ignore the first transient screw-axis samples when choosing the measured
# "best" axis for comparison and gain suggestion. Among stable samples, the
# code picks the axis with the smallest distance to the active contact edge.
# If no sample passes this stable-window gate, the code still keeps the best
# measured transient fallback and labels it as transient_fallback.
post_contact_best_axis_min_time = 0.60
post_contact_best_axis_min_omega = 0.12

# Moment change that means the tool has settled/made firm contact, for
# example the second tool edge touching down after passively tipping flat
# [Nm]. Early-exit threshold for post_contact_align; only checked once
# post_contact_align_min_time has elapsed.
# Raised because with the much higher preload force now in use, the steady
# preload alone was already producing ~30 Nm at only 9.8 deg of actual tip --
# the old 25 Nm threshold was firing before the tool finished flattening.
post_contact_moment_threshold = 60.0

# Small position offset into the real surface during post_contact_align [m].
# This acts like a spring preload, pressing the selected edge down:
#   preload force approx post_contact_Kp_z * post_contact_normal_push
# Starts higher so a real downward force is felt immediately at phase entry,
# not only after several seconds of ramp.
post_contact_normal_push = 0.018

# During post_contact_align, keep pushing slightly deeper into the surface
# over time. This grows the preload/contact moment at the pressed edge,
# which is what should passively tip the tool flat. Total push is clamped by
# post_contact_max_push.
post_contact_push_speed = 0.010
post_contact_max_push = 0.031

# Translational spring used only during post_contact_align, in the base
# frame [x0, y0, z0]. z0 (downward, into the table) is intentionally stiff
# so the preload above generates a real contact moment at the pressed edge:
# 5500 N/m * 0.018 m = 99 N immediately, ramping to 5500 N/m * 0.031 m =
# 170.5 N peak. Real Cartesian safety limit is collision_force_acc/nom
# (200 N, see below) -- collision_cartesian_force_threshold (45 N) is NOT
# actually enforced by this controller. 170.5 N peak leaves ~29.5 N margin.
# Medium x0/y0 stiffness for the sideways-compliance pole test. The direct
# one-sample suggested-gain test caused a reflex stop at post_align t=0.1 s
# because it combined high stiffness with almost zero damping, so these gains
# are kept as the safer validated post-contact baseline.
