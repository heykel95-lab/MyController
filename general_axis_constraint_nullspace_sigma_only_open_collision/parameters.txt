# ------------------------------------------------------------------
# Spatial surface constraint mode
# ------------------------------------------------------------------
# axis_constraint_mode = 1 and use_surface_constraint = 1:
#
# The constrained surface is a plane:
#
#   n^T * (p - p_surface) = 0
#
# surface_normal defines n in the robot base frame.
# use_start_as_surface_point = 1 uses the start TCP position as p_surface.
# This is safest because the surface is never far from the initial robot pose.
#
# The impedance gains are chosen in the surface frame:
#
#   x = surface normal direction
#   y = first tangent direction
#   z = second tangent direction
#
# Then the code transforms them to the robot base frame:
#
#   K_base = R_surface * K_surface * R_surface^T
#
# Example Y-Z plane:
#   surface_normal = [1, 0, 0]
#
# Example inclined plane:
#   surface_normal = [0.7071, 0.0, 0.7071]
axis_constraint_mode = 1
use_surface_constraint = 1
use_start_as_surface_point = 1

surface_point_x = 0.0

# ------------------------------------------------------------------
# Automatic contact search for the virtual surface point
# ------------------------------------------------------------------
# use_contact_search = 1:
#   Before the surface impedance starts, the TCP moves slowly in
#   contact_search_direction until contact is detected.
#
# Contact detection uses the estimated external force only as a trigger:
#
#   ||F_ext - F_ext_start|| > contact_force_threshold
#
# This is not force control. After the trigger, the current TCP position
# becomes the virtual surface point:
#
#   surface_point_runtime = p_contact
#
# Keep disabled until the search direction and distance are checked safely.

# Example downward search for a horizontal surface.

# Slow search speed [m/s] and maximum travel distance [m].

# Contact trigger threshold [N] using force change from start.

# Softer Cartesian gains used only during contact search.
