#
# Fix two axes:
#   fix_p_x=1, fix_p_y=1, fix_p_z=0 -> motion mainly along z
#
# Fix all axes:
#   fix_p_x=1, fix_p_y=1, fix_p_z=1 -> return-to-point impedance
