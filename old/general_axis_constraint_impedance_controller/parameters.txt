# ------------------------------------------------------------------
# General axis constraint mode
# ------------------------------------------------------------------
# axis_constraint_mode = 1:
#   For translation:
#     fix_p_i = 1 -> p_d_i = p_start_i, axis is fixed/constrained
#     fix_p_i = 0 -> p_d_i = p_EE_i, axis is free in position
#
# Examples:
#   Y-Z plane: fix_p_x=1, fix_p_y=0, fix_p_z=0
#   X-Z plane: fix_p_x=0, fix_p_y=1, fix_p_z=0
#   X-Y plane: fix_p_x=0, fix_p_y=0, fix_p_z=1
#
# You can also fix two axes:
#   fix_p_x=1, fix_p_y=1, fix_p_z=0 -> motion mainly along z
#
# Or fix all axes:
#   fix_p_x=1, fix_p_y=1, fix_p_z=1 -> normal return-to-point impedance
axis_constraint_mode = 1

# Default: Y-Z plane, x fixed
fix_p_x = 1
fix_p_y = 0
fix_p_z = 0

# ------------------------------------------------------------------
# Rotation axis constraints
# ------------------------------------------------------------------
# The orientation error vector e_R is computed normally.
# Then each component can be kept or released:
#   fix_R_i = 1 -> keep rotational stiffness in that component
#   fix_R_i = 0 -> set e_R_i=0, no rotational spring in that component
#
# Damping still acts according to DR_i.
# For a fully free rotational component, also set the corresponding DR_i = 0.
