# ====================================================================
# HOLD mode gains
# ====================================================================
# Used when the startup menu picks hold instead of the phase sequence (or when
# use_phase_sequence = 0). Hold locks the captured start pose: POSITION with
# hold_Kp/hold_Dp and ORIENTATION with hold_KR/hold_DR, both isotropic in the
# base frame. The auto-damping toggle/factor for hold live in common.txt.
#
# From hold you can press g+Enter to hand-guide the tool and p+Enter to restart
# the whole sequence from the new pose.

# Cartesian position stiffness [N/m], base x/y/z.
hold_Kp = 300.0

# Manual damping FALLBACK [Ns/m], paired with hold_Kp. Used only when
# hold_auto_damping is off, OR when it is on but the online task-space inertia
# estimate is unavailable at hold entry -- i.e. near a kinematic singularity
# (Jacobian loses rank) or if the inertia comes out non-finite / non-positive.
# Otherwise D = hold_auto_damping_factor * 2 * sqrt(M * hold_Kp) is used.
hold_Dp = 50.0

# Isotropic rotational hold spring [Nm/rad | Nms/rad]. hold_DR is applied
# directly -- hold rotation is NOT auto-damped.
hold_KR = 40.0
hold_DR = 8.0
