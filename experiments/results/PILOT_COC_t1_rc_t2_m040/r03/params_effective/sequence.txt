# ====================================================================
# PHASE SEQUENCE parameters
# ====================================================================
# Sequence run: approach -> set up -> grind.
# Surface/tool geometry and auto-damping toggles live in common.txt.

# 1 = sequence. 0 = hold at the start pose.
use_phase_sequence = 1

# 0 = skip orient and start at descend.
use_approach_orient = 1


# ====================================================================
# Phase 1: approach
# ====================================================================
# ----- orient step -----
# Minimum time in the orient step before it may switch to descend [s].
approach_orient_min_time = 0.5

# Tool-axis error threshold for switching to descend [rad].
approach_orient_error_threshold = 0.035

# ----- shared approach impedance (orient AND descend) -----
# Surface-frame order: [tangent1, tangent2, normal].
approach_Kp_normal = 150.0
approach_Kp_tangent1 = 150.0
approach_Kp_tangent2 = 150.0
approach_KR_normal = 90.0
approach_KR_tangent1 = 90.0
approach_KR_tangent2 = 90.0
# Manual damping fallback [Ns/m | Nms/rad].
approach_Dp_normal = 20.0
approach_Dp_tangent1 = 20.0
approach_Dp_tangent2 = 20.0
approach_DR_normal = 12.0
approach_DR_tangent1 = 12.0
approach_DR_tangent2 = 12.0

# ----- descend step -----
# Switches to set up when the active contact point reaches this clearance.
descend_surface_clearance = 0.020

# Descend moves into the surface along -normal.
# Speed [m/s] and maximum travel before the run gives up [m].
descend_speed = 0.1
descend_max_distance = 0.640


# ====================================================================
# Phase 2: set up
# ====================================================================
# Ends on setup_moment_threshold after setup_min_time, or on setup_timeout.
# 12.0 from G2: the push ramp ends at 4.0 s, force overshoots to ~65 N and
# relaxes to ~60 N by 4.6 s, after which tip and force are flat to 12 s
# (t8 drift 0.002 deg, t12 drift 0.000 deg). At 4.0 s the phase was being cut
# at the overshoot, so every number taken from it was a transient.
setup_min_time = 2.0
setup_timeout = 5.0
setup_moment_threshold = 150.0

# Alignment starts from the actual contact-point height captured at the end of
# descend. End + speed define the ramp; ramp time is derived as
# abs(end - captured_start) / speed. setup_timeout is independent and may stop
# the phase before the configured end is reached.
# Check the active normal stiffness times setup_push_end against the configured
# collision/reflex thresholds before every new experimental range.
setup_push_speed = 0.050
setup_push_end = 0.060

# 0 = legacy base diagonal below. 1 = surface [tangent1,tangent2,normal].
setup_translation_surface_frame = 1

# Legacy translational spring, base frame [x, y, z].
setup_Kp_x = 2000.0
setup_Kp_y = 2000.0
setup_Kp_z = 350.0
# Manual damping FALLBACK [Ns/m], same fallback conditions as the approach Dp.
setup_Dp_x = 10.0
setup_Dp_y = 10.0
setup_Dp_z = 175.0

# Surface-frame translational spring used by the calibrated experiments.
setup_Kp_surface_tangent1 = 2000.0
setup_Kp_surface_tangent2 = 2000.0
setup_Kp_surface_normal = 800.0
# Manual damping fallback in the same frame. Automatic damping replaces these
# values when valid task inertia is available.
setup_Dp_surface_tangent1 = 50.0
setup_Dp_surface_tangent2 = 50.0
setup_Dp_surface_normal = 25.0

# Rotational spring, surface frame [tangent1, tangent2, normal].
setup_KR_normal = 50.0
setup_KR_tangent1 = 5.0
setup_KR_tangent2 = 5.0
# Manual damping FALLBACK [Nms/rad].
setup_DR_normal = 10.0
setup_DR_tangent1 = 10.01
setup_DR_tangent2 = 10.01


# ====================================================================
# Phase 3: grind
# ====================================================================
# 1 = commanded sweep. 0 = free-slide press hold.
grind_sweep_enabled = 0
print_grind_debug = 0

# Sweep path: smooth ping-pong along tangent1 or tangent2.
grind_axis = 2
grind_amplitude_m = 0.03
grind_frequency_hz = 0.2


# ====================================================================
# Enter gates between phases
# ====================================================================
# Bare Enter continues; e+Enter stops.
pause_before_set_up = 1
pause_before_grind = 0

# Stiff hold while paused at the pre-set-up gate. Auto damping fits translation
# toward pause_hold_Dp and rotation toward the approach damping inherited by the
# old gate behavior.
pause_hold_Kp = 5000.0
pause_hold_Dp = 200.0
pause_hold_auto_damping = 1


# ====================================================================
# Coupled (pole-based) stiffness during set up
# ====================================================================
# Command the 6x6 spring during set up.
use_coupled_stiffness = 1

# K/D source: block diagonal or a deliberately commanded pole.
coupled_use_block_diagonal = 0
coupled_pole_manual = 1

# Preferred experiment convention. When enabled, these entries define exactly
# r_c = p_TCP - p_c in the surface frame [tangent1,tangent2,normal].
coupled_use_direct_rc_surface = 1
coupled_rc_tangent1 = 0.0
coupled_rc_tangent2 = -0.040
coupled_rc_normal = 0.0

# Legacy base-frame pole-from-edge convention, retained for archived setups.
coupled_pole_from_edge_x = -0.04
coupled_pole_from_edge_y = 0.08
coupled_pole_from_edge_z = 0.0

# 1 = freeze pole at first contact. 0 = update from the live edge.
coupled_pole_freeze_at_contact = 1

# Print the coupled-stiffness diagnostics in the set-up report.
print_coupled_diagnostics = 1
